import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from scipy import stats

data = {
    'StudentId': range(1,11),
    'MathsScore' : [85,92,np.nan,78,88,85,79,93,105,74],
    'EnglishScore' : [78,85,90,82,np.nan,88,74,91,86,80],
    'ScienceScore' : [89,91,94,76,88,np.nan,72,85,98,74]
}

df = pd.DataFrame(data)

print("\n Initial Dataset: \n")
print(df)

#Check for missing Values:
missing_values = df.isnull().sum()
print("\n Missing Values in each column: \n")
print(missing_values)

#Handle missing values by filling it with the meean on col
df['MathsScore']=df["MathsScore"].replace(np.nan,df["MathsScore"].mean())
df['EnglishScore']=df["EnglishScore"].replace(np.nan,df["EnglishScore"].mean())
df['ScienceScore'] = df["ScienceScore"].replace(np.nan, df["ScienceScore"].mean())
print('\nDataset after handling missing values: \n')
print(df)

#Boxplot to visually inspect outliers
plt.figure(figsize=(10,6))
sns.boxplot(data=df[['MathsScore','EnglishScore','ScienceScore']])
plt.title("\nBoxplot to detect outliers\n")
plt.show()

#Detect outliers with Z-score
z_scores = np.abs(stats.zscore(df[['MathsScore','EnglishScore','ScienceScore']]))
print("\n Z-scores of dataset: \n")
print(z_scores)

#Set a threshold for z-score
outliers = (z_scores > 3)
print("\nOutliers detected (True means outlier): \n")
print(outliers)

#Apply log transformation to MathsScore to decrease skewness
df['MathsScore_log'] = np.log(df['MathsScore'] + 1)

#visualizing transform effect
plt.figure(figsize = (10,6))
sns.histplot(df['MathsScore'], kde=True, color='orange', label='Original MathsScore')
sns.histplot(df['MathsScore_log'], kde=True, color='red', label='Log-Transformed MathsScore')
plt.legend()
plt.title("Effect of Log Transformation on MathsScore")
plt.show()

#Display final transformed dataset
print("\n Final transformed Dataset: \n")
print(df)
