import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline

# Load the Boston Housing dataset
data = pd.read_csv("https://raw.githubusercontent.com/selva86/datasets/master/BostonHousing.csv")

# Display the first and last few rows
print("First few rows of the data:")
print(data.head())
print("\nLast few rows of the data:")
print(data.tail())

# Display the shape of the data
print("\nThe shape of the data is: ")
print(data.shape)

# Check for missing values
print("\nMissing values in each column:")
print(data.isnull().sum())

# Split data into features (X) and target (y)
X = data.iloc[:, 0:13]  # Features
y = data.iloc[:, -1]    # Target (last column)

# Split the dataset into training and testing sets (80% train, 20% test)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.20, random_state=42)

# Print shapes of training and test sets
print("\nShape of training and test sets:")
print(f"X_train shape: {X_train.shape}")
print(f"X_test shape: {X_test.shape}")
print(f"y_train shape: {y_train.shape}")
print(f"y_test shape: {y_test.shape}")

# Create a pipeline with StandardScaler (scaling features) and LinearRegression model
model = make_pipeline(StandardScaler(), LinearRegression())

# Fit the model on the training data
model.fit(X_train, y_train)

# Evaluate the model on the test data
score = model.score(X_test, y_test)
print(f"\nModel R-squared on test data: {score:.4f}")
