import pandas as pd
from sklearn import preprocessing

# Load the dataset with proper column names
csv_url = 'https://archive.ics.uci.edu/ml/machine-learning-databases/iris/iris.data'
col_names = ['Sepal_Length', 'Sepal_Width', 'Petal_Length', 'Petal_Width', 'Species']
iris = pd.read_csv(csv_url, header=None, names=col_names)

# Displaying the first 8 rows and the last few rows
print(iris.head(8))
print(iris.tail())

# Checking the index, columns, shape, and data types
print(iris.index)
print(iris.columns)
print(iris.shape)
print(iris.dtypes)

# Summary statistics of the dataset
print(iris.describe())

# Accessing specific rows and columns
print(iris.iloc[5])  # Row 5
print(iris[47:51])  # Rows from 47 to 51
print(iris.loc[:, ["Sepal_Length", "Sepal_Width"]])  # Specific columns

# Accessing columns 2 to 4 (Petal_Length and Petal_Width)
cols_2_4 = iris.columns[2:4]
print(iris[cols_2_4])

# Checking for missing values
print(iris.isnull().any())
print(iris.isnull().sum())

# Changing Petal_Length to integer
iris['Petal_Length'] = iris['Petal_Length'].astype("int")

# Min-Max scaling of the features
min_max_scaler = preprocessing.MinMaxScaler()
X = iris.iloc[:, :4]  # Selecting the feature columns
X_scaled = min_max_scaler.fit_transform(X)
df_normalized = pd.DataFrame(X_scaled, columns=iris.columns[:4])

# Encoding the 'Species' column with one-hot encoding
enc = preprocessing.OneHotEncoder()
enc_df = enc.fit_transform(iris[['Species']]).toarray()
enc_df = pd.DataFrame(enc_df, columns=['Iris-Setosa', 'Iris-Versicolor', 'Iris-Virginica'])

# Joining the encoded 'Species' column back with the features
df_encode = iris.drop(columns=['Species'])
df_encode = df_encode.join(enc_df)

# Displaying the final encoded DataFrame
print(df_encode)
