import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Load the dataset
dataset = pd.read_csv('https://raw.githubusercontent.com/shivang98/Social-Network-ads-Boost/master/Social_Network_Ads.csv')

# Show the first few rows of the dataset
print("First few rows of the dataset:")
print(dataset.head())

# Show dataset information (columns, types, non-null values)
print("\nDataset Information:")
print(dataset.info())

# Extract features and target variable
X = dataset.iloc[:, [2, 3]].values  # Age and EstimatedSalary columns
y = dataset.iloc[:, 4].values  # Purchased column (0 = No, 1 = Yes)

# Print first few feature and target values
print("\nFirst 3 feature values:")
print(X[:3, :])
print('-'*15)
print("First 3 target values:")
print(y[:3])

# Show the last few rows of the dataset
print("\nLast few rows of the dataset:")
print(dataset.tail())

# Split dataset into training and test sets (25% test, 75% train)
from sklearn.model_selection import train_test_split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size = 0.25, random_state = 0)

# Print first few train and test feature and target values
print("\nFirst 3 feature values in training set:")
print(X_train[:3])
print('-'*15)
print("First 3 target values in training set:")
print(y_train[:3])
print('-'*15)
print("First 3 feature values in test set:")
print(X_test[:3])
print('-'*15)
print("First 3 target values in test set:")
print(y_test[:3])

# Feature scaling (standardize features)
from sklearn.preprocessing import StandardScaler
sc_X = StandardScaler()
X_train = sc_X.fit_transform(X_train)
X_test = sc_X.transform(X_test)

# Print first few standardized feature values
print("\nFirst 3 standardized feature values in training set:")
print(X_train[:3])
print('-'*15)
print("First 3 standardized feature values in test set:")
print(X_test[:3])

# Logistic Regression classifier
from sklearn.linear_model import LogisticRegression
classifier = LogisticRegression(random_state = 0, solver='lbfgs')
classifier.fit(X_train, y_train)

# Predict on the test set
y_pred = classifier.predict(X_test)

# Print first 10 predicted and actual values
print("\nFirst 10 predicted values:")
print(y_pred[:10])
print('-'*15)
print("First 10 actual values:")
print(y_test[:10])

# Print first 20 predicted and actual values
print("\nFirst 20 predicted values:")
print(y_pred[:20])
print("First 20 actual values:")
print(y_test[:20])

# Confusion Matrix to evaluate the model performance
from sklearn.metrics import confusion_matrix
cm = confusion_matrix(y_test, y_pred)

# Print confusion matrix
print("\nConfusion Matrix:")
print(cm)

# Optionally, print the accuracy score
from sklearn.metrics import accuracy_score
accuracy = accuracy_score(y_test, y_pred)
print(f"\nAccuracy Score: {accuracy:.4f}")
