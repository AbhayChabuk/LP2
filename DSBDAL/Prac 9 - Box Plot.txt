import seaborn as sns
import matplotlib.pyplot as plt

# Load Titanic dataset
titanic = sns.load_dataset("titanic")

# Display first 10 rows
print(titanic.head(10))

# Get info about the dataset
print(titanic.info())

# Get descriptive statistics for the dataset
print(titanic.describe())

# Display columns 'survived' and 'alive'
print(titanic.loc[:, ["survived", "alive"]])

# Boxplot of 'age' vs 'sex'
plt.figure(figsize=(10, 5))
sns.boxplot(x="sex", y="age", data=titanic)
plt.title('Boxplot: Age vs Sex')
plt.show()

# Boxplot of 'age' vs 'sex' with hue 'survived'
plt.figure(figsize=(10, 5))
sns.boxplot(x="sex", y="age", data=titanic, hue="survived")
plt.title('Boxplot: Age vs Sex with Hue Survived')
plt.show()

# Violinplot of 'age' vs 'sex'
plt.figure(figsize=(10, 5))
sns.violinplot(x="sex", y="age", data=titanic)
plt.title('Violinplot: Age vs Sex')
plt.show()

# Violinplot of 'age' vs 'sex' with hue 'survived'
plt.figure(figsize=(10, 5))
sns.violinplot(x="sex", y="age", data=titanic, hue="survived", split=True)
plt.title('Violinplot: Age vs Sex with Hue Survived')
plt.show()
